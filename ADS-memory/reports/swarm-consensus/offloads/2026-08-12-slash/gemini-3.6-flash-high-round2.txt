- **Ranking Criteria:**

  2. Strict adherence to package neutrality ([slots.ts:77-80](file:///tmp/ads-peer-dispatch/debate-slash/files/jini/slots.ts#L77-L80)).
  3. Transactional safety (async promise handling and draft recovery on host failure).
  4. Backward compatibility with standard prompt macros.

- **Genuine Sacrifice:** The package does not auto-execute side effects out-of-the-box. Every executable command requires the host to handle argument resolution and execution in `onDiscoverySelect`.
- **Cheapest Falsifying Test:** Type `/search quantum computing` into the draft and press Enter. If the palette closes without passing `"quantum computing"` to `onDiscoverySelect`, or if typing a space after `/search ` immediately dismisses the palette and erases the draft, Option 1 is falsified.

- **Mechanism:** `@jini-ai/chat` incorporates a built-in execution engine. `ComposerDiscoveryItem` defines explicit `executionType: 'macro' | 'client-route' | 'agent-tool'`, and `Composer.tsx` branches internally to invoke client routes or trigger agent tool calls directly.

// Update in file:///tmp/ads-peer-dispatch/debate-slash/files/jini/composer-discovery.ts

import type { ComposerDiscoveryGroup, ComposerDiscoveryItem } from '../slots.js';

export interface ParsedSlashQuery {
  /** Command name used for palette filtering, e.g. "mcp" from "/mcp supabase" */
  commandQuery: string;
  /** Argument text typed after command, e.g. "supabase" from "/mcp supabase" */
  rawArgs: string;
  /** True if whitespace has been entered after the command name */
  hasSpace: boolean;
  /** Start and end character indices of the active slash trigger in the draft */
  range: readonly [number, number];
}

/** 
 * N1 Parser: Matches slash commands at draft start with optional argument text.
 * Replaces rigid /^\/([^\s/]*)$/ anchored regex.
 */
export function parseComposerSlashQuery(draft: string): ParsedSlashQuery | null {
  const match = draft.match(/^\/([^\s/]*)(?:\s+(.*))?$/);
  if (!match) return null;

  const commandQuery = match[1] ?? '';
  const rawArgs = match[2] ?? '';
  const hasSpace = draft.length > 1 + commandQuery.length && draft[1 + commandQuery.length] === ' ';

  return {
    commandQuery,
    rawArgs,
    hasSpace,
    range: [0, draft.length],
  };
}

/** Flattens non-empty groups while preserving host order, then applies case-insensitive search. */
export function filterComposerDiscovery(
  groups: readonly ComposerDiscoveryGroup[],
  query: string,
): ComposerDiscoveryMatch[] {
  const normalizedQuery = query.trim().toLowerCase();
  const matches: ComposerDiscoveryMatch[] = [];

  for (const group of groups) {
    for (const item of group.items) {
      const searchable = [item.label, item.description, item.kind, ...(item.keywords ?? [])]
        .filter((value): value is string => typeof value === 'string')
        .join(' ')
        .toLowerCase();
      if (normalizedQuery === '' || searchable.includes(normalizedQuery)) {
        matches.push({ groupId: group.id, groupLabel: group.label, item });
      }
    }
  }

  return matches;
}

/** 
 * Replaces ONLY the active trigger range rather than doing destructive full-draft replacement.
 * Fixes coupled invariant bug flagged in N1 coordinator finding.
 */
export function replaceComposerSlashTrigger(
  draft: string,
  insertText: string,
  range?: readonly [number, number],
): string {
  if (!range) return parseComposerSlashQuery(draft) === null ? draft : insertText;
  return draft.slice(0, range[0]) + insertText + draft.slice(range[1]);
}

// Modifications in file:///tmp/ads-peer-dispatch/debate-slash/files/jini/Composer.tsx

export function Composer({
  composer,
  onSend,
  disabled = false,
  sendDisabled = false,
  placeholder,
  slots,
  attachmentPicker,
  running = false,
  onCancel,
}: ComposerProps) {
  const t = useT();
  const attachmentInputRef = useRef<HTMLInputElement | null>(null);
  const textareaRef = useRef<HTMLTextAreaElement | null>(null);
  const [discoveryMenuOpen, setDiscoveryMenuOpen] = useState(false);
  const [slashActiveIndex, setSlashActiveIndex] = useState(0);
  const [dismissedSlashDraft, setDismissedSlashDraft] = useState<string | null>(null);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const resolvedPlaceholder = placeholder ?? t('Send a message…');
  const discoveryGroups = slots?.discoveryGroups ?? [];
  const hasDiscoveryItems = discoveryGroups.some((group) => group.items.length > 0);
  
  // N1 Parser Integration
  const parsedSlash = parseComposerSlashQuery(composer.draft);
  const slashQuery = parsedSlash?.commandQuery ?? null;
  const slashMatches = slashQuery === null ? [] : filterComposerDiscovery(discoveryGroups, slashQuery);
  const slashOpen = slashMatches.length > 0 && dismissedSlashDraft !== composer.draft;

  function restoreComposerFocus() {
    textareaRef.current?.focus();
  }

  // N1 & N3: Async Intent Notification with Transactional Draft Rollback
  async function notifyDiscovery(
    item: ComposerDiscoveryItem,
    source: 'plus' | 'slash',
    rawArgs?: string,
  ) {
    if (!slots?.onDiscoverySelect) return;
    const originalDraft = composer.draft;
    try {
      setErrorMessage(null);
      await slots.onDiscoverySelect({ item, source, rawArgs });
    } catch (err) {
      // Transactional rollback on host execution rejection
      composer.setDraft(originalDraft);
      setErrorMessage(err instanceof Error ? err.message : t('Command execution failed'));
    }
  }

  async function selectSlashItem(index: number) {
    const match = slashMatches[index];
    if (!match) return;

    // Replace trigger using exact range
    const insertValue = match.item.insertText ?? match.item.label;
    const nextDraft = replaceComposerSlashTrigger(composer.draft, insertValue, parsedSlash?.range);
    composer.setDraft(nextDraft);
    setDismissedSlashDraft(null);

    await notifyDiscovery(match.item, 'slash', parsedSlash?.rawArgs);
    restoreComposerFocus();
  }

  function selectPlusItem(item: ComposerDiscoveryItem) {
    if (item.insertText) composer.setDraft(appendComposerDiscovery(composer.draft, item.insertText));
    setDiscoveryMenuOpen(false);
    void notifyDiscovery(item, 'plus');
    restoreComposerFocus();
  }

  function handleKeyDown(event: KeyboardEvent<HTMLTextAreaElement>) {
    if (event.nativeEvent.isComposing) return;
    if (slashOpen) {
      const action = resolveComposerSlashKeyAction(event.key, event.shiftKey);
      if (action.type !== 'none') {
        event.preventDefault();
        if (action.type === 'move') {
          setSlashActiveIndex((current) => (current + action.offset + slashMatches.length) % slashMatches.length);
        } else if (action.type === 'select') {
          void selectSlashItem(Math.min(slashActiveIndex, slashMatches.length - 1));
        } else {
          setDismissedSlashDraft(composer.draft);
        }
        return;
      }
    }
    if (event.key !== 'Enter' || event.shiftKey) return;
    event.preventDefault();
    if (!disabled && !sendDisabled && composer.canSubmit) onSend();
  }

  const activeItem = slashMatches[Math.min(slashActiveIndex, slashMatches.length - 1)]?.item;

  return (
    <div className="jini-composer">
      {errorMessage ? <div className="jini-composer-error" role="alert">{errorMessage}</div> : null}
      {slots?.leadingAccessories ? <div className="jini-composer-leading">{slots.leadingAccessories}</div> : null}
      <AttachmentTray attachments={composer.attachments} onRemove={composer.removeAttachment} />
      <textarea
        ref={textareaRef}
        className="jini-composer-input"
        value={composer.draft}
        placeholder={resolvedPlaceholder}
        disabled={disabled}
        onChange={(e) => {
          composer.setDraft(e.target.value);
          setSlashActiveIndex(0);
          setDismissedSlashDraft(null);
          setErrorMessage(null);
        }}
        onKeyDown={handleKeyDown}
        aria-controls={slashOpen ? 'jini-composer-slash-menu' : undefined}
        aria-expanded={slashOpen}
        aria-activedescendant={
          slashOpen
            ? `jini-composer-slash-option-${Math.min(slashActiveIndex, slashMatches.length - 1)}`
            : undefined
        }
        rows={3}
      />
      {slashOpen ? (
        <div className="jini-composer-slash-container">
          <ComposerSlashMenu
            matches={slashMatches}
            activeIndex={Math.min(slashActiveIndex, slashMatches.length - 1)}
            onSelect={(item) => void selectSlashItem(slashMatches.findIndex((match) => match.item === item))}
            t={t}
          />
          {/* N3: Host Slot Preview Flyout */}
          {activeItem && slots?.renderDiscoveryPreview ? (
            <div className="jini-composer-preview-flyout">
              {slots.renderDiscoveryPreview(activeItem)}
            </div>
          ) : null}
        </div>
      ) : null}
      {/* ... footer controls ... */}
    </div>
  );
}

## Strongest Counter-Argument

The strongest counter-argument to Option 1 is:

> *"Delegating argument resolution and execution handling entirely to the host (`onDiscoverySelect`) forces every host application to write manual dispatch boilerplate for common capabilities (such as client navigation, prompt expansion, or agent tool triggers). A package-managed execution engine (Option 2) would eliminate host glue code by standardizing execution types directly inside `@jini-ai/chat`."*

### Rebuttal
This counter-argument fails on three critical architectural requirements:
1. **Violation of Written Design Laws ([slots.ts:77-80](file:///tmp/ads-peer-dispatch/debate-slash/files/jini/slots.ts#L77-L80)):** Hardcoding execution types (`'client-route'`, `'agent-tool'`, `'plugin'`) inside `@jini-ai/chat` couples a generic React UI package to Tovu’s internal runtime architecture.
2. **Failure under N4 (Unknown Future Kinds):** The owner explicitly mandated extensibility to unknown future capability kinds. A package-managed engine breaks whenever a host introduces a novel capability lifecycle that does not match the package's hardcoded dispatch matrix.
3. **Authorization & Security Boundaries:** Execution of consequential commands requires host-specific consent frameworks (such as Tovu's `PendingConfirmationStore` at [AssistantDock.tsx:44-73](file:///tmp/ads-peer-dispatch/debate-slash/files/tovu/AssistantDock.tsx#L44-L73)). The package cannot know whether a command requires admin session credentials, daemon proxy validation, or two-phase redemption. Keeping dispatch in `onDiscoverySelect` preserves strict security separation.

## What Would Change My Mind

I would reconsider Option 2 or a package-managed execution model under two specific empirical conditions:
1. **Cross-Host Standardized RPC Protocol:** If all present and future consumer applications of `@jini-ai/chat` agreed on a single, standardized client-agent RPC tool execution protocol (e.g. standardized JSON-RPC schemas over WebSockets), standardizing execution dispatch inside the package would become viable without breaking reusability.
2. **Strict Text-Macro Scoping:** If product requirements rolled back N1 and N3—explicitly forbidding command arguments, side-effect execution, and live previews—restricting slash commands strictly to static text insertion would render host delegation unnecessary.
