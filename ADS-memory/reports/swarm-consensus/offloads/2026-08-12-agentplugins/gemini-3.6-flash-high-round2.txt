### 1. `files/tovu/capability-projection.ts`
Defines the core `CapabilityDescriptor` type, argument schemas, execution bindings, preview provider interface, trust provenance, and `CapabilityRegistry` (satisfying N1, N2, N3, N4).

```typescript
import type { ComposerDiscoveryGroup, ComposerDiscoveryItem } from "../jini/composer-discovery.js";

/** N3: Extensible capability kind string. Predefined kinds or arbitrary future kinds. */
export type CapabilityKind = "skill" | "mcp" | "agent-tool" | "plugin" | "connector" | (string & {});

/** N4: Typed argument descriptor for palette rendering and execution validation. */
export interface CapabilityArgument {
  readonly name: string;
  readonly type: "string" | "number" | "boolean" | "json";
  readonly description: string;
  readonly required: boolean;
  readonly placeholder?: string;
  readonly defaultValue?: unknown;
}

/** N4: Polymorphic execution binding contract. Executed by the host engine, never by vendor JS. */
export type CapabilityExecuteBinding =
  | { readonly type: "context-assembly"; readonly promptText: string }
  | {
      readonly type: "tool-dispatch";
      readonly toolId: string; // Authority-qualified ID (e.g. mcp__conn1__tool)
      readonly transformArgs?: (raw: Record<string, unknown>) => Record<string, unknown>;
    }
  | { readonly type: "route-navigation"; readonly route: string }
  | {
      readonly type: "custom-handler";
      readonly handler: (args: Record<string, unknown>) => Promise<{ success: boolean; result: unknown }>;
    };

/** N1: Generic preview payload structure, replacing static allowlists. */
export interface CapabilityPreviewFile {
  readonly relativePath: string;
  readonly mimeType?: string;
}

export interface CapabilityPreviewContent {
  readonly title: string;
  readonly defaultFilePath?: string;
  readonly files: readonly CapabilityPreviewFile[];
  readonly getFileContent: (relativePath: string) => Promise<string | null>;
}

export interface CapabilityPreviewResolver {
  readonly hasPreview: boolean;
  readonly resolvePreview: () => Promise<CapabilityPreviewContent | null>;
}

/** Trust & Provenance classification matching Tovu trust posture. */
export interface CapabilityProvenance {
  readonly source: "bundled" | "agent-plugin" | "native-plugin" | "federated-mcp" | "agent-core";
  readonly providerId: string; // e.g. "ui-ux-design"
  readonly providerVersion: string;
  readonly trustStatus: "trusted-native" | "context-only" | "admitted-mcp" | "unadmitted-mcp";
}

/** The unified master descriptor contract for all host capabilities. */
export interface CapabilityDescriptor {
  readonly id: string; // Global unique identifier
  readonly kind: CapabilityKind;
  readonly label: string;
  readonly description: string;
  readonly keywords?: readonly string[];
  readonly icon?: string;
  readonly arguments?: readonly CapabilityArgument[];
  readonly execute: CapabilityExecuteBinding;
  readonly preview?: CapabilityPreviewResolver;
  readonly provenance: CapabilityProvenance;
  readonly confirmationRequired: boolean;
  readonly groupCategory: string; // Grouping bucket for composer UI
}

/** N3: Generic capability provider interface. Sources implement this without touching composer code. */
export interface CapabilityProvider {
  readonly providerId: string;
  getDescriptors(): Promise<readonly CapabilityDescriptor[]>;
}

/** N2: Live Capability Registry feeding AssistantDock.tsx and Jini composer. */
export class CapabilityRegistry {
  private readonly providers = new Map<string, CapabilityProvider>();

  public registerProvider(provider: CapabilityProvider): void {
    this.providers.set(provider.providerId, provider);
  }

  public unregisterProvider(providerId: string): void {
    this.providers.delete(providerId);
  }

  public async getAllDescriptors(): Promise<readonly CapabilityDescriptor[]> {
    const results: CapabilityDescriptor[] = [];
    for (const provider of this.providers.values()) {
      const descriptors = await provider.getDescriptors();
      results.push(...descriptors);
    }
    return results;
  }

  /** Projects live descriptors into Jini ComposerDiscoveryGroup format for AssistantDock.tsx. */
  public async projectToComposerGroups(): Promise<readonly ComposerDiscoveryGroup[]> {
    const descriptors = await this.getAllDescriptors();
    const groupMap = new Map<string, ComposerDiscoveryItem[]>();

    for (const desc of descriptors) {
      const groupKey = desc.groupCategory;
      if (!groupMap.has(groupKey)) {
        groupMap.set(groupKey, []);
      }

      // Convert typed arguments to syntax placeholder hint (e.g. /search <query>)
      const argHint = desc.arguments?.map((a) => (a.required ? `<${a.name}>` : `[${a.name}]`)).join(" ") ?? "";
      const displayLabel = argHint ? `${desc.label} ${argHint}` : desc.label;

      const item: ComposerDiscoveryItem = {
        id: desc.id,
        label: displayLabel,
        description:
          desc.provenance.trustStatus === "unadmitted-mcp"
            ? `[Pending Authorization] ${desc.description}`
            : desc.description,
        kind: desc.kind,
        keywords: desc.keywords ? [...desc.keywords] : undefined,
        insertText: desc.execute.type === "context-assembly" ? desc.execute.promptText : `/${desc.label} `,
      };

      groupMap.get(groupKey)!.push(item);
    }

    const groups: ComposerDiscoveryGroup[] = [];
    for (const [groupCategory, items] of groupMap.entries()) {
      groups.push({
        id: groupCategory.toLowerCase().replace(/\s+/g, "-"),
        label: groupCategory,
        items,
      });
    }

    return groups;
  }
}
```

### 2. `files/tovu/agent-plugin-adapter.ts`
Adapts Agent Plugins (`plugin.json`, `skills/*/SKILL.md`, `mcp.json`) into `CapabilityDescriptor` objects. Handles optional presentation-only `extensions["org.tovu.commands"]`, `$schema` mismatches, collision resolution, and path-sanitized previews.

```typescript
import type {
  CapabilityArgument,
  CapabilityDescriptor,
  CapabilityProvider,
} from "./capability-projection.js";

export interface AgentPluginRawFile {
  readonly relativePath: string;
  readonly content: string;
}

export interface AgentPluginPackage {
  readonly manifestJson: string;
  readonly files: readonly AgentPluginRawFile[];
}

/** Presentation metadata schema allowed in extensions["org.tovu.commands"]. NO EXECUTABLE CODE. */
export interface TovuCommandPresentationMeta {
  readonly label?: string;
  readonly description?: string;
  readonly icon?: string;
  readonly groupCategory?: string;
  readonly arguments?: readonly {
    readonly name: string;
    readonly type: "string" | "number" | "boolean" | "json";
    readonly description: string;
    readonly required: boolean;
    readonly placeholder?: string;
  }[];
}

export class AgentPluginCapabilityAdapter implements CapabilityProvider {
  public readonly providerId: string;
  private readonly pkg: AgentPluginPackage;
  private readonly admittedMcpTools: ReadonlySet<string>;

  constructor(
    pluginId: string,
    pkg: AgentPluginPackage,
    admittedMcpTools: ReadonlySet<string> = new Set()
  ) {
    this.providerId = pluginId;
    this.pkg = pkg;
    this.admittedMcpTools = admittedMcpTools;
  }

  public async getDescriptors(): Promise<readonly CapabilityDescriptor[]> {
    const descriptors: CapabilityDescriptor[] = [];
    let manifest: Record<string, unknown> = {};

    try {
      manifest = JSON.parse(this.pkg.manifestJson);
    } catch {
      return []; // Malformed manifest produces zero descriptors
    }

    const pluginId = (manifest.name as string) || this.providerId;
    const version = (manifest.version as string) || "1.0.0";
    const extensions = (manifest.extensions as Record<string, unknown>) || {};
    const commandMetaMap = (extensions["org.tovu.commands"] as Record<string, TovuCommandPresentationMeta>) || {};

    // 1. Process Skills (skills/<name>/SKILL.md)
    for (const file of this.pkg.files) {
      const skillMatch = file.relativePath.match(/^skills\/([^/]+)\/SKILL\.md$/);
      if (skillMatch) {
        const skillName = skillMatch[1];
        const canonicalId = `agent-plugin:${pluginId}:skill:${skillName}`;
        const meta = commandMetaMap[`skill:${skillName}`] || {};

        descriptors.push({
          id: canonicalId,
          kind: "skill",
          label: meta.label || skillName,
          description: meta.description || `Portable skill '${skillName}' from ${pluginId}`,
          keywords: ["skill", skillName, pluginId],
          icon: meta.icon,
          arguments: meta.arguments as CapabilityArgument[] | undefined,
          execute: {
            type: "context-assembly",
            promptText: file.content,
          },
          preview: this.createPreviewResolver(skillName, file.relativePath),
          provenance: {
            source: "agent-plugin",
            providerId: pluginId,
            providerVersion: version,
            trustStatus: "context-only",
          },
          confirmationRequired: false,
          groupCategory: meta.groupCategory || "Skills",
        });
      }
    }

    // 2. Process MCP Servers (mcp.json) with $schema mismatch handling
    const mcpFile = this.pkg.files.find((f) => f.relativePath === "mcp.json");
    if (mcpFile) {
      try {
        const mcpConfig = JSON.parse(mcpFile.content);

        // Validate mcp.json $schema if present
        if (mcpConfig.$schema && typeof mcpConfig.$schema === "string") {
          if (!mcpConfig.$schema.includes("modelcontextprotocol")) {
            console.warn(`[AgentPlugin] $schema mismatch in mcp.json for ${pluginId}. Skipping MCP tools.`);
          } else {
            this.parseMcpServers(pluginId, version, mcpConfig, commandMetaMap, descriptors);
          }
        } else {
          this.parseMcpServers(pluginId, version, mcpConfig, commandMetaMap, descriptors);
        }
      } catch (err) {
        // $schema parse error invalidates MCP config while skills remain loaded (graceful isolation)
        console.error(`[AgentPlugin] Failed to parse mcp.json for ${pluginId}:`, err);
      }
    }

    return descriptors;
  }

  private parseMcpServers(
    pluginId: string,
    version: string,
    mcpConfig: Record<string, unknown>,
    metaMap: Record<string, TovuCommandPresentationMeta>,
    descriptors: CapabilityDescriptor[]
  ): void {
    const servers = (mcpConfig.mcpServers as Record<string, Record<string, unknown>>) || {};

    for (const [serverName, serverSpec] of Object.entries(servers)) {
      const declaredTools = (serverSpec.tools as readonly string[]) || [];

      for (const toolName of declaredTools) {
        const qualifiedToolId = `mcp__${pluginId}_${serverName}__${toolName}`;
        const isAdmitted = this.admittedMcpTools.has(qualifiedToolId);
        const meta = metaMap[`mcp:${serverName}:${toolName}`] || {};

        descriptors.push({
          id: `agent-plugin:${pluginId}:mcp:${serverName}:${toolName}`,
          kind: "mcp",
          label: meta.label || `${serverName}:${toolName}`,
          description: meta.description || `MCP tool '${toolName}' from server '${serverName}'`,
          keywords: ["mcp", serverName, toolName, pluginId],
          icon: meta.icon,
          arguments: meta.arguments as CapabilityArgument[] | undefined,
          execute: {
            type: "tool-dispatch",
            toolId: qualifiedToolId,
          },
          provenance: {
            source: "agent-plugin",
            providerId: pluginId,
            providerVersion: version,
            trustStatus: isAdmitted ? "admitted-mcp" : "unadmitted-mcp",
          },
          confirmationRequired: !isAdmitted,
          groupCategory: meta.groupCategory || "MCP Tools",
        });
      }
    }
  }

  /** Creates path-sanitized PreviewResolver for N1 generic previewing. */
  private createPreviewResolver(skillName: string, mainSkillPath: string) {
    const files = this.pkg.files;
    return {
      hasPreview: true,
      resolvePreview: async () => ({
        title: `Skill: ${skillName}`,
        defaultFilePath: mainSkillPath,
        files: files.map((f) => ({ relativePath: f.relativePath })),
        getFileContent: async (requestedPath: string) => {
          // Strict path traversal prevention: reject '..' or non-relative paths
          if (requestedPath.includes("..") || requestedPath.startsWith("/")) {
            return null;
          }
          const target = files.find((f) => f.relativePath === requestedPath);
          return target ? target.content : null;
        },
      }),
    };
  }
}
```

```typescript
import type { CapabilityDescriptor, CapabilityPreviewContent } from "./capability-projection.js";

export class GenericCapabilityPreviewProvider {
  /** Resolves preview payload for any capability descriptor without knowing its underlying kind. */
  public async getPreview(descriptor: CapabilityDescriptor): Promise<CapabilityPreviewContent | null> {
    if (!descriptor.preview || !descriptor.preview.hasPreview) {
      return null;
    }
    return descriptor.preview.resolvePreview();
  }

  /** Validates path containment to prevent directory traversal in preview requests. */
  public static sanitizeRelativePath(basePath: string, relativePath: string): string | null {
    const normalized = relativePath.replace(/\\/g, "/");
    if (normalized.includes("../") || normalized.includes("/..") || normalized.startsWith("/")) {
      return null;
    }
    return normalized;
  }
}
```

## The Trust Gap

The fundamental trust problem in Agent Plugins carrying `mcp.json` is the **authorship mismatch**:
- `mcp-federation`'s Rule R2 ([`trust.ts:65-69`](file:///tmp/ads-peer-dispatch/debate-agentplugins/files/tovu/mcp-federation/trust.ts#L65-L69)) explicitly states:
  > *"A remote tool is admitted only if the site owner's `allowedToolNames` names it. Discovery does not confer availability. This is the direct structural analogue of `DerivedRiskByToolId`: the classification that decides admission is authored by someone other than the thing being classified."*
- A plugin's `mcp.json` is authored by the plugin developer (an untrusted remote party).
- Rule R3 ([`trust.ts:70-74`](file:///tmp/ads-peer-dispatch/debate-agentplugins/files/tovu/mcp-federation/trust.ts#L70-L74)) reinforces:
  > *"SELF-DECLARED HINTS DEMOTE ONLY, NEVER PROMOTE... The untrusted party is given exactly one power over its own privileges — the power to reduce them."*

### How Option 1 Solves the Gap Without Compromise
1. **Default Unadmitted Posture:** Tools declared inside a plugin's `mcp.json` are ingested into `mcp-federation` under authority-qualified namespaces ([`trust.ts:57-63`](file:///tmp/ads-peer-dispatch/debate-agentplugins/files/tovu/mcp-federation/trust.ts#L57-L63)), but are assigned `trustStatus = "unadmitted-mcp"` and `confirmationRequired = true`.
2. **Explicit Site Owner Admission:** Stdio MCP tools cannot execute until the Tovu site administrator explicitly adds the qualified tool ID (`mcp__<pluginId>_<server>__<tool>`) to `allowedToolNames` in workspace settings.
3. **Execution Gate:** Attempting to execute an unadmitted MCP command triggers a blocking UI authorization modal requiring site owner credentials before dispatching to `mcp-federation`.
4. **Honesty Gate Disclosure:** In the `AgentPlugins.tsx` UI and composer descriptions, Skills report `Status: Ready (Context Assembly)`, while stdio MCP tools report `Status: Pending Site Owner Authorization`.

## What Would Change My Mind

1. **Upstream Standard Command Specification:** Ratification of an official, normative `/commands` schema in `agent-plugins.org` v1.1+ with cryptographic signature verification (removing the need for presentation-only extensions).
2. **OS Subprocess Sandbox Landing in Host:** Implementation of an OS-level sandbox runner (e.g. Landlock, AppArmor, or lightweight OCI containers) inside Tovu for stdio subprocesses, making unvetted stdio MCP tools safe to execute without explicit operator allowlisting.
3. **Third-Party Security Attestation Registry:** Broad industry adoption of an independent, cryptographically signed security attestation network for MCP servers.
