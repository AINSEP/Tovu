import { buildCapabilityScopedSdk } from "./capability-sdk";
import type { PluginCapability } from "./manifest";
import type { ContentEntryDraft, PluginSdk } from "../../../packages/sdk/src/index";

export interface LoadPluginOptional {
  readonly runtimeSdkVersion?: string;
  readonly importModule?: (entryPath: string) => Promise<unknown>;
  readonly computeFileHash?: (absoluteFilePath: string) => Promise<string>;
  /** Live hook registry instance for step (5) attachment. */
  readonly hookRegistry?: HookRegistry;
  /** Current workspace ID for per-load SDK closing. */
  readonly workspaceId?: string;
  /** Data accessors for capability SDK stubs. */
  readonly getCurrentEntry?: () => Readonly<ContentEntryDraft>;
  readonly writeExtField?: (field: string, value: string | number | boolean) => void;
}

export async function loadPlugin(
  required: LoadPluginRequired,
  _optional: LoadPluginOptional = {}
): Promise<LoadPluginResult> {
  const { manifest, entryPath, record } = required;
  const importModule = _optional.importModule ?? ((p: string) => import(p));
  const computeFileHash = _optional.computeFileHash ?? defaultComputeFileHash;
  const runtimeSdkVersion = _optional.runtimeSdkVersion ?? DEFAULT_RUNTIME_SDK_VERSION;

  // --- Step (1): Integrity Check ---
  const pluginRoot = derivePluginRoot(entryPath);
  for (const [relativeFilePath, expectedHash] of Object.entries(manifest.integrity)) {
    let actualHash: string;
    try {
      actualHash = await computeFileHash(path.join(pluginRoot, relativeFilePath));
    } catch {
      return { loaded: false, reason: "INTEGRITY_FAILED" };
    }
    if (actualHash !== expectedHash) {
      return { loaded: false, reason: "INTEGRITY_FAILED" };
    }
  }

  // --- Step (2): SDK Range Check ---
  if (!semver.satisfies(runtimeSdkVersion, manifest.sdkRange)) {
    return { loaded: false, reason: "SDK_RANGE_UNSATISFIED" };
  }

  // --- Step (3): ESM Dynamic Import ---
  let importedModule: unknown;
  try {
    importedModule = await importModule(entryPath);
  } catch {
    return { loaded: false, reason: "CODE_ENTRY_MISSING" };
  }

  // --- Steps (4) & (5): Build Capability SDK, Call setup(), Attach Hooks ---
  if (_optional.hookRegistry && typeof importedModule === "object" && importedModule !== null) {
    const mod = importedModule as { setup?: (sdk: PluginSdk) => void | Promise<void> };
    let capturedFilter: BeforeSaveFilter | undefined;

    const sdk = buildCapabilityScopedSdk({
      pluginId: manifest.id,
      capabilities: manifest.capabilities as PluginCapability[],
      coreDeps: {
        getCurrentEntry: _optional.getCurrentEntry ?? (() => ({ id: "", workspaceId: _optional.workspaceId ?? "", title: "", body: "", ext: {} })),
        writeExtField: _optional.writeExtField ?? (() => {}),
        attachFilter: (hookName, filter) => {
          if (hookName === "content.entry.beforeSave") {
            capturedFilter = filter;
          }
        },
      },
    });

    if (typeof mod.setup === "function") {
      await mod.setup(sdk);
    }

    if (capturedFilter) {
      const declaredFields: HookRegistryFieldDecl[] = manifest.fields.map((f) => ({
        path: f.path,
        type: f.type,
      }));
      attachLoadedPlugin({
        pluginId: manifest.id,
        source: record.source === "built-in" ? "built-in" : "site",
        hookRegistry: _optional.hookRegistry,
        filter: capturedFilter,
        declaredFields,
      });
    }
  }

  return { loaded: true };
}
```

export async function setPluginEnabled(
  required: SetPluginEnabledRequired,
  _optional: SetPluginEnabledOptional = {}
): Promise<{ activation: PluginActivationRecord }> {
  const { deps, input } = required;
  const discovered = deps.discovery.find((record) => record.id === input.pluginId);
  if (!discovered) {
    throw new PluginNotFoundError(`plugin '${input.pluginId}' was not found in discovery snapshot`);
  }

  if (input.enabled) {
    if (discovered.status === "incompatible") {
      throw new PluginIncompatibleError(`plugin '${input.pluginId}' is incompatible`);
    }
    if (discovered.status !== "valid") {
      throw new PluginInvalidError(`plugin '${input.pluginId}' failed validation`);
    }
  }

  const existing = await deps.repo.getActivation({ workspaceId: input.workspaceId, pluginId: input.pluginId });
  const version = input.enabled ? discovered.version : (existing?.version ?? discovered.version);

  const activation: PluginActivationRecord = {
    pluginId: input.pluginId,
    workspaceId: input.workspaceId,
    version,
    enabled: input.enabled,
    updatedAt: deps.clock.nowIso(),
  };

  // Persist state transition
  await deps.repo.save(activation);

  if (input.enabled) {
    try {
      await deps.onEnabled?.(input.pluginId);
    } catch (error) {
      // Revert database activation state if setup() or loader throws
      const reverted: PluginActivationRecord = {
        ...activation,
        enabled: false,
        updatedAt: deps.clock.nowIso(),
      };
      await deps.repo.save(reverted);
      throw error;
    }
  } else {
    deps.onDisabled?.(input.pluginId);
  }

  return { activation };
}
```

### 2. Auto-Quarantine & ABI Deep Isolation (`hook-registry.ts`)

```typescript
import type { JsonObject } from "@jini-ai/cms/core";
import type { BeforeSaveFilter, ContentEntryDraft } from "../../../packages/sdk/src/index";

export interface QuarantineHandler {
  (pluginId: string, error: PluginHookFailedError): Promise<void>;
}

export interface HookRegistry {
  attach(
    pluginId: string,
    source: AttachmentSource,
    filter: BeforeSaveFilter,
    declaredFields: readonly HookRegistryFieldDecl[]
  ): void;
  detach(pluginId: string): void;
  runBeforeSave(entry: Readonly<ContentEntryDraft>): Promise<JsonObject>;
  setQuarantineHandler(handler: QuarantineHandler): void;
}

interface Attachment {
  readonly source: AttachmentSource;
  readonly filter: BeforeSaveFilter;
  readonly declaredFields: readonly HookRegistryFieldDecl[];
  consecutiveFailures: number;
}

const FAILURE_THRESHOLD = 3;

export function createHookRegistry(): HookRegistry {
  const attachments = new Map<string, Attachment>();
  let quarantineHandler: QuarantineHandler | undefined;

  function attach(
    pluginId: string,
    source: AttachmentSource,
    filter: BeforeSaveFilter,
    declaredFields: readonly HookRegistryFieldDecl[]
  ): void {
    attachments.set(pluginId, { source, filter, declaredFields, consecutiveFailures: 0 });
  }

  function detach(pluginId: string): void {
    attachments.delete(pluginId);
  }

  function setQuarantineHandler(handler: QuarantineHandler): void {
    quarantineHandler = handler;
  }

  async function runBeforeSave(entry: Readonly<ContentEntryDraft>): Promise<JsonObject> {
    const ordered = [...attachments.entries()].sort(compareTb01);
    const merged: Record<string, JsonObject> = {};

    for (const [pluginId, attachment] of ordered) {
      // ABI Audit Fix: Deep clone to enforce memory boundary isolation (MDN structuredClone)
      const baseSnapshot: ContentEntryDraft = structuredClone(entry);
      const snapshot: ContentEntryDraft = {
        ...baseSnapshot,
        ext: { ...baseSnapshot.ext, ...merged },
      };
      const ctx = { pluginId, workspaceId: entry.workspaceId };

      let patch: unknown;
      try {
        // ABI Audit Fix: Explicitly await asynchronous filter Promise<ExtPatch>
        patch = await attachment.filter(snapshot, ctx);
        attachment.consecutiveFailures = 0; // Reset counter on success
      } catch (error) {
        attachment.consecutiveFailures += 1;
        const hookError = new PluginHookFailedError(
          pluginId,
          `plugin '${pluginId}' filter failed (${attachment.consecutiveFailures}/${FAILURE_THRESHOLD}): ${
            error instanceof Error ? error.message : String(error)
          }`,
          { cause: error }
        );

        if (attachment.consecutiveFailures >= FAILURE_THRESHOLD && quarantineHandler) {
          detach(pluginId); // Immediate circuit breaker detach
          await quarantineHandler(pluginId, hookError); // Auto-disable gateway REQ-07
        }
        throw hookError;
      }

      if (typeof patch !== "object" || patch === null || Array.isArray(patch)) {
        throw new PluginHookFailedError(pluginId, `plugin '${pluginId}' returned non-object ext patch`);
      }

      const declaredByField = new Map(attachment.declaredFields.map((f) => [fieldNameOf(f.path), f]));
      const pluginPatch: Record<string, JsonObject[string]> = {};

      for (const [field, value] of Object.entries(patch as Record<string, unknown>)) {
        const decl = declaredByField.get(field);
        if (!decl) {
          throw new PluginHookFailedError(pluginId, `plugin '${pluginId}' returned undeclared ext field '${field}'`);
        }
        if (!matchesDeclaredType(value, decl.type)) {
          throw new PluginHookFailedError(pluginId, `plugin '${pluginId}' field '${field}' type mismatch`);
        }
        pluginPatch[field] = value as JsonObject[string];
      }

      merged[pluginId] = pluginPatch;
    }

    return merged;
  }

  return { attach, detach, runBeforeSave, setQuarantineHandler };
}
```

## Critique Of Another Participant's Round 2 Code

### Critique 1: `gemini-3.6-flash-high-round2` (Activation State Atomicity Defect)
In `gemini-3.6-flash-high-round2` (lines 313–316), the proposed enablement sequence states:
> *"setPluginEnabled() ... must be wired to pass a composition-root onEnabled handler that: (a) executes loadPlugin(), (b) passes capability SDK to setup(), and (c) calls attachLoadedPlugin()."*

**Contract Defect:** `files/plugin-runtime/activation.ts:127-130` saves the database row (`await deps.repo.save(activation)`) **before** invoking `await deps.onEnabled?.(input.pluginId)`. If `loadPlugin()`, module evaluation, or `setup()` throws an exception (e.g., dynamic import error or failed integrity check), `setPluginEnabled()` propagates the error, but the `plugin_activations` row remains written as `enabled: true`. On subsequent server boots, `getActivation()` reports the plugin as enabled even though setup failed. The code must include a transactional `try/catch` rollback in `setPluginEnabled()` to revert the activation record on failure.

### Critique 2: `sonnet5-round2` (Circuit Breaker State Leak)
In `sonnet5-round2` (lines 255–260), the proposed `onDisabled` callback is described as:
> *"onDisabled: (id) => deps.hookRegistry.detach(id)"*

**Contract Defect:** Detaching a plugin removes its entry from `attachments.delete(pluginId)`, but if circuit breaker state (consecutive error count) is stored in a external map or retained across re-attaches, re-enabling a previously failing plugin immediately inherits stale failure counters. `attach()` must reset `consecutiveFailures: 0` explicitly so that manual operator re-enablement via `PATCH .../plugins/:id {enabled: true}` provides a clean slate.
