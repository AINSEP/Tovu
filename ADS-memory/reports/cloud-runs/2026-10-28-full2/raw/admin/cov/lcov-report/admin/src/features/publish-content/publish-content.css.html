/**
 * `features/publish-content/PublishContentDialog.tsx` — the publish report table and its dialog width.
 *
 * Every colour, radius and spacing value below is an existing token from `styles.css`
 * (`--surface*`, `--border`, `--muted`, `--warning`, `--space-*`, `--text-*`), so the table
 * inherits light and dark mode from the same place the rest of the app does and this file declares
 * no palette of its own.
 *
 * The width override follows `external-mcp-tool-picker.css`'s precedent for widening the shared
 * `.settings-dialog` (26rem by default, which is right for a two-sentence confirm and far too narrow
 * for a five-column report): `width` AND `max-width`, because `.settings-dialog-backdrop` is
 * `display: grid; place-items: center` and a `max-width` alone leaves the dialog shrink-wrapped to
 * its text. Applied via `.publish-content-dialog`, which the component only adds once a report
 * exists — the dialog opens at its normal confirm-dialog width and widens when there is a table to
 * hold.
 */
 
.settings-dialog.publish-content-dialog {
  width: min(48rem, 92vw);
  max-width: min(48rem, 92vw);
  /* The dialog itself is capped, not just the table inside it. Without this the modal grew to its
     content's height and filled a laptop window edge to edge (owner, 2026-09-19); `.settings-dialog`
     is already `display: flex; flex-direction: column`, so a cap here is all the scrolling region
     below needs to become the one thing that gives. */
  max-height: min(42rem, 84vh);
}
 
/* The report can be long (a first publish plans every post, page and media row on the site), so it
   scrolls inside the dialog rather than growing the dialog past the viewport and stranding the
   Cancel/Publish buttons below the fold. */
.publish-content-report {
  /* The ONLY region in this dialog that scrolls. `flex: 1 1 auto` lets it take whatever the capped
     dialog has left after the heading, the two lines of copy and the button row; `min-height: 0` is
     what actually permits it to shrink below its content (a flex item's default `min-height: auto`
     refuses to, which is exactly how an "overflow: auto" child ends up pushing its parent past the
     viewport instead of scrolling). `max-height` stays as a floor-independent cap for the case where
     the dialog is NOT height-constrained — a short plan in a tall window keeps its old shape. */
  flex: 1 1 auto;
  min-height: 6rem;
  max-height: min(30rem, 52vh);
  overflow-y: auto;
  border-radius: var(--r-lg);
}
 
/* publish-content-copy-2026-09-25.md — this table's own six columns were overflowing the dialog:
   `.list-table`'s shared `table-layout: auto` (`styles.css`) sizes every column from its content, and
   with two free-text columns (Entity, Why) competing for room the browser grew the table past the
   dialog's own capped width, squeezed "Why" to an unreadable strip, and pushed "Overwrite on live"
   against the dialog's right edge with nothing to clip it back (owner, 2026-09-25, `.table-scroll`'s
   own horizontal scrollbar never became visible enough to notice, so it read as a bug, not a scroll).
   `table-layout: fixed` sizes every column from its `<thead>` cell's own width — the table algorithm
   always reads the FIRST row, so a width has to live on the `<th>`, not the `<td>`, to have any effect
   — and gives whichever column declares no width at all (now "What happens", which carries each row's
   reason under its status since the Why column went, owner 2026-09-26) the entire remainder.
   Every width here is a PERCENTAGE of the table's own rendered width rather than a fixed `rem`, so the
   five fixed columns and the one flexible one shrink together in proportion at 390px mobile instead of
   the fixed-`rem` columns eating a bigger and bigger share of an ever-narrower table. Scoped to `.publish-content-report .list-table`, never bare `.list-table`,
   so none of the other ~15 admin screens sharing that class lose their own `auto` content-sizing. */
.publish-content-report .list-table {
  table-layout: fixed;
}
 
/* Owner-facing plain words ("Post", "Redirect", "Theme" — `report-rows.ts`'s `ENTITY_TYPE_LABEL`)
   run a little longer than the raw type ids this column was first sized for; 10% clipped "Redirect"
   to "Redire…" (owner report, 2026-09-25). */
.publish-content-report .list-table th:nth-child(2) {
  width: 14%;
}
 
/* The type column is a short word ('Post', 'Media', 'Theme') and should not claim a share of
   the row; ellipsis is a defensive floor, not the expected case — every real value fits. */
.publish-content-report td:nth-child(2) {
  font-size: var(--text-2xs);
  color: var(--muted);
  white-space: nowrap;
  overflow: hidden;
  text-overflow: ellipsis;
}
 
/* The Item column got part of the old Why column's room: names were cut off at 22% (owner
   screenshot 2026-09-26). The rest goes to "What happens", which now carries the reasons. */
.publish-content-report .list-table th:nth-child(3) {
  width: 30%;
}
 
/* The entity column now carries a slug or a title, not a uuid — monospace and `break-all` were
   there to make an opaque id scannable and would only make readable words harder to read. The column
   width (set on the `<th>` above — see this file's header) stops one long slug from taking the reason
   column's room; `overflow`/`text-overflow` here still do the actual per-cell clipping. */
.publish-content-report td:nth-child(3) {
  overflow: hidden;
  text-overflow: ellipsis;
  white-space: nowrap;
}
 
/* The checkbox column: as narrow as the control, centred, and never a click target wider than it
   looks. A percentage (not the old fixed `rem`) so it shrinks with every other column at 390px rather
   than holding its full desktop width while the flexible column gives up all of its own — see this
   file's header. */
.publish-content-report .publish-content-select {
  width: 8%;
  padding-right: 0;
  text-align: center;
  vertical-align: middle;
}
 
/* The "Overwrite on live" column: only rendered while at least one row in the current plan offers it
   (`PublishContentDialog.tsx`'s `showOverwriteColumn`), so it costs no width the rest of the time. */
.publish-content-report .publish-content-overwrite {
  width: 14%;
  white-space: normal;
}
 
.publish-content-report .publish-content-select input {
  margin: 0;
  cursor: pointer;
}
 
.publish-content-report .publish-content-select input:disabled {
  cursor: default;
}
 
/* A row's reason, under its status pill in the "What happens" cell — the only text in this dialog an
   operator has to actually read, so it wraps rather than truncating. */
.publish-content-reason {
  display: block;
  margin-top: var(--space-1);
  font-size: var(--text-xs);
  color: var(--muted);
  white-space: normal;
}
 
/* A skipped row is the one thing the operator must not scroll past. A left rail in the existing
   warning token marks it without recolouring the whole row, which would fight the `.status-warning`
   pill already in its own cell. */
.publish-content-report tbody tr[data-publish-disposition="skipped"] td:first-child {
  box-shadow: var(--jini-status-warning-shadow);
}
 
.publish-content-report tbody tr[data-publish-disposition="unchanged"] td {
  color: var(--muted);
}
 
/* The "N already up to date" fold row (owner 2026-09-26): a quiet link-button with a caret that turns
   when the rows it hides are showing. */
.publish-content-up-to-date td {
  font-size: var(--text-sm);
}
 
.publish-content-up-to-date .link-button::before {
  content: "▸";
  display: inline-block;
  width: 1em;
  font-size: 1.15em;
  transition: transform 0.16s ease;
}
 
.publish-content-up-to-date .link-button[aria-expanded="true"]::before {
  transform: rotate(90deg);
}
 
.publish-content-summary {
  margin: 0;
  font-size: var(--text-2xs);
}
 
.publish-content-target {
  margin: 0;
  font-size: var(--text-sm);
}
