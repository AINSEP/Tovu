/* Jini owns the native modal lifecycle. These CMS dialogs keep their labelled
   domain actions, like Jini MediaPicker, so a second generic Close adds no action.
   `.jini-domain-dialog` is the host-neutral name Jini admin dialogs use (Jini 47b9fa4f, shipped
   in @jini-ai/admin 0.4.11); Tovu's own dialogs use it too. The old host-named alias was
   dropped once nothing emitted it (2026-10-14). */
.jini-domain-dialog > .jini-dialog-close { display: none; }
.settings-dialog.jini-domain-dialog {
  border: 0; color: var(--fg); margin: auto;
  /* CustomSelect portals into this native dialog with viewport-fixed coordinates. Keep its
     shell unfiltered; the independently filtered ::backdrop supplies the frosted background. */
  /* Native dialogs shrink to their content without an explicit width; the media
     instance editor's short labels used to squeeze its fields into 230px. */
  width: var(--admin-dialog-width, 32rem);
  max-width: calc(100vw - 2rem);
  box-sizing: border-box;
}
.jini-domain-dialog:not([open]) { display: none; }
.jini-domain-dialog::backdrop { background: var(--jini-overlay-bg);
  backdrop-filter: var(--tovu-admin-scrim-backdrop, none);
}
/* The native title sits outside each form; the form still owns submission and
   keeps the same vertical rhythm the former form-as-dialog root supplied. */
.jini-domain-dialog > form { display: flex; flex-direction: column; gap: 0.9rem; }
 
/* Native Dialog owns modal focus/cancellation. Its body/footer composition keeps long forms
   scrollable while the title and actions stay available, without a second dialog lifecycle. */
.jini-domain-dialog[open]:has(> form > .jini-dialog-body) {
  display: flex;
  flex-direction: column;
  max-height: min(85vh, calc(100dvh - var(--space-8)));
  overflow: hidden;
}
.jini-domain-dialog:has(> form > .jini-dialog-body) > form {
  min-height: 0;
  overflow: hidden;
}
/* The body borrows the dialog's own padding (negative margin, matching padding) so a scrolling
   body's overlay scrollbar sits beside the fields, not over their right edge and eye toggles, and
   the scroll box no longer clips their focus rings. Content width is unchanged. */
.jini-domain-dialog .jini-dialog-body {
  display: flex;
  flex-direction: column;
  gap: 0.9rem;
  min-height: 0;
  overflow-y: auto;
  margin: -3px calc(-1 * var(--space-3));
  padding: 3px var(--space-3);
}
.jini-domain-dialog .jini-dialog-footer {
  display: flex;
  flex-shrink: 0;
  justify-content: flex-end;
  gap: var(--space-2);
  padding-top: var(--space-3);
  border-top: 1px solid var(--border);
}
.jini-domain-dialog .jini-dialog-section-title {
  margin: 0 0 var(--space-3);
  font-family: var(--font-display);
  font-size: var(--text-md);
  font-weight: 700;
  color: var(--fg);
}
