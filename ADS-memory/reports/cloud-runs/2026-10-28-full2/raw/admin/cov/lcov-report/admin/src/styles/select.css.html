/**
 * `components/Select/Select.tsx` — trigger + floating panel for the custom searchable dropdown
 * (`WidgetPickerDialog.tsx`'s two selects only; every other `<select>` in this admin stays native,
 * see that component's own header). Imported directly by `Select.tsx`, the same scoped-partial
 * precedent `styles/assistant.css` established (see `AssistantDock.tsx`'s import and that file's
 * own header) rather than a new line in `main.tsx`.
 *
 * The floating panel (`.select-panel`) is a real `document.body` portal, not an in-place
 * absolutely-positioned child — two problems confirmed live, not assumed:
 *   1. `.widget-picker-body` (added by this same dispatch's Task 1) sets `overflow-y: auto` on the
 *      dialog's scrolling body. An in-place `position: absolute` panel is a normal descendant of
 *      that scroll container and would get clipped by it the moment the panel needed to render
 *      past the visible scroll area. `position: fixed`, portaled out of that subtree, is measured
 *      against the viewport instead and is not clipped by any ancestor's `overflow`.
 *   2. `.settings-dialog-backdrop` sets `position: fixed; z-index: 80`, which establishes a
 *      stacking context. A `position: fixed` DESCENDANT of that backdrop still paints INSIDE that
 *      stacking context regardless of its own z-index — CSS gives no way for a normal descendant to
 *      escape an ancestor's stacking context except leaving that subtree entirely. A real portal to
 *      `document.body` does that; a same-subtree `position: fixed` child would not.
 * `Select.tsx`'s own `computePosition`/scroll-reposition logic keeps this panel anchored to the
 * trigger regardless of where in the DOM it physically renders.
 */
 
/* Trigger — sized to match `input, select` (`styles.css`) so it sits flush next to a real text
   input on the same row/form; this is what stands in for a native `<select>`, so it must share
   that control's metrics, not just its color.
 
   CORRECTION: this used to claim `.select-trigger` won against the app-wide bare `button` rule
   because this file's `<style>` tag is injected after `styles.css`'s, at equal (0,1,0) specificity.
   Both halves of that were wrong, found live rather than assumed, after a second agent working on
   an unrelated Forms-editor dialog in this same tree hit the opposite outcome — their own
   self-imported partial's `width` LOST to `.settings-dialog`'s `max-width` from `styles.css`, so
   "a component's self-imported CSS lands after `styles.css`" is not a reliable mechanism at all
   (it depends on when in the render tree that component's module first evaluates — this app has
   route-level code that doesn't all load eagerly at boot — not on static import order in the
   source). Checked THIS file's actual DOM order via `document.styleSheets` while the dialog was
   open: `select.css`'s `<style>` tag is index 0, `styles.css`'s is index 1 — this file loads
   FIRST, the opposite of the original claim. It wins anyway, because the specificity claim was
   also wrong: `button` alone is a bare type selector, specificity (0,0,1), not (0,1,0) — `.select-
   trigger` (a class selector, (0,1,0)) beats it unconditionally, regardless of which stylesheet
   loads first. Verified via `getComputedStyle` in the real browser: `border-color` and `padding`
   both come from this rule, not the bare `button` rule, independent of load order. Every other
   selector below that shares a target with `styles.css` (`:hover`, `:focus-visible`, `:disabled`)
   is in the same shape — a class(+pseudo-class) selector against a bare-element(+pseudo-class)
   one — so all of them win on specificity alone too; none of this file's rules actually depend on
   its `<style>` tag's position in the document. */
.select-trigger {
  width: 100%; box-sizing: border-box;
  display: inline-flex; align-items: center; justify-content: space-between; gap: var(--space-2);
  background: var(--jini-control-bg); color: var(--fg);
  /* An accent-tinted resting border, not the plain `--border-strong` every real input/select gets —
     the reference this component was built against shows the trigger itself carrying that color
     at rest (not just on focus), which is also the one visible cue that this control opens a
     search panel rather than an OS popup. `color-mix` keeps it a tint of the token, not the full
     saturated `--primary`, so it doesn't read as an error/required-field state next to plain
     inputs in the same form.
 
     `in oklab`, not `in oklch` — found live, not a style preference. `--border-strong` sits at hue
     255° (blue), `--primary` at 43.4° (rust); `color-mix(in oklch, ...)` interpolates hue around
     the wheel by the SHORTER arc, and the shorter path from 255° to 43.4° (148.4° vs the 211.6° the
     other way) runs backward through 300s°/0°, i.e. straight through magenta. At the 45% this
     rule originally used, that landed at hue 321.78° — a cool lavender, not a tint of `--primary`
     at all, and the user correctly flagged it against a rust reference. Raising the percentage
     doesn't escape this: it only walks further along the same magenta-crossing arc (measured:
     70% → hue 358.9°, still dusty rose; 80% → hue 13.7°, still not `--primary`'s own 43.4°; only
     ~90%+ gets hue-close to `--primary`, at which point it is barely a tint anymore). `oklab`
     blends the `a`/`b` channels linearly instead of walking the hue wheel, so there is no polar
     detour — at this SAME 45% ratio it resolves to hue ≈39.9°, almost exactly `--primary`'s own
     hue, sampled as `#c09c90` (a warm terracotta) via `getComputedStyle`/canvas pixel read in a
     real browser. Generalizable trap: mixing toward any cool-hued neutral (`--border`,
     `--border-strong`, `--surface`'s own neutral variants) with a warm accent in `oklch` risks
     the same magenta detour whenever the two hues are far enough apart that the "shorter" arc
     crosses 0°/360° — `oklab` (or `in oklch longer hue`, less predictable) is the fix, not a
     bigger ratio. */
  border: var(--jini-border-width, 1px) solid color-mix(in oklab, var(--primary) 45%, var(--border-strong));
  border-radius: var(--r-sm); padding: 0.55rem 0.7rem; min-height: 37px;
  font-family: var(--font-ui); font-size: var(--text-sm); font-weight: 400; letter-spacing: normal;
  cursor: pointer; text-align: left;
  transition: border-color 0.14s, box-shadow 0.14s;
  box-shadow: var(--jini-control-shadow, none);
}

/* Selected row — the same low-alpha `color-mix` tint idiom `styles.css` uses for tinted fills
   (`color-mix(in oklab, var(--primary) 30%, var(--surface))` here)
   rather than a flat `--primary-dim` background, so this reads as "the primary color, faded" and
   not a generic separate tint. Text goes fully `--primary` (not mixed) — the row still needs to
   read clearly at 13px, and the tinted background alone already carries the "faded" quality. */
/* `in oklab`, not `in oklch`, for the same reason that rule uses it (its comment has the full
   mechanism): oklch interpolates HUE along the shorter arc, and dark mode's `--surface` is
   `oklch(18.5% 0.018 255)` — real blue chroma, not powerless — so mixing `--primary` (hue 44)
   with it travelled the short way round through purple AND gained chroma on the way. Measured
   in dark before the change: `C 0.0295 H 275.9` here and `C 0.0360 H 287.8` on the highlighted
   row — visibly violet selected rows, more saturated than the surface they sit on. OKLab is the
   same perceptual space in rectangular a/b coordinates, so there is no arc to take the wrong
   way: the small warm contribution now partly CANCELS the surface's blue instead of rotating
   past it, measuring `C 0.0080` and `C 0.0123` after — below the level at which a hue is
   identifiable, i.e. the intended "surface, very slightly tinted". Light mode is untouched
   (`--surface` is `oklch(100% 0 none)`, and a `none` component is carried forward in ANY
   interpolation space): bit-identical `L 0.9374 C 0.0158 H 43.4` / `L 0.9016 C 0.0248 H 43.4`
   before and after, which is why neither rule needs a dark-scoped arm. */

  color: var(--primary);
  font-weight: 600;
}
