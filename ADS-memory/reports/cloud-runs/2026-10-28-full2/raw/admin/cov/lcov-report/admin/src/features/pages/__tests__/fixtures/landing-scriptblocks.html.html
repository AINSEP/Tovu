  /* Proof deck playback. kUInetic moves the deck; this only plays and stops the videos.
     Play on the centre slide plays inline with native controls. Play on a peeking neighbour
     centres it first (via its dot), then plays. A slide that leaves the centre, a video that
     ends, or the deck scrolling out of view all put the slide back to its poster + button. */
  (function () {
    'use strict';
    var deck = document.querySelector('.tv [data-od-id="proof-deck"]');
    if (!deck) { return; }
    var slides = Array.prototype.slice.call(deck.querySelectorAll('.deck-slide'));
    var dots = Array.prototype.slice.call(deck.querySelectorAll('.deck-dot'));
    var videoOf = function (slide) { return slide.querySelector('video'); };
    var stop = function (slide) {
      var video = videoOf(slide);
      if (!video || !slide.hasAttribute('data-playing')) { return; }
      video.pause();
      video.controls = false;
      slide.removeAttribute('data-playing');
    };
    var play = function (slide) {
      var video = videoOf(slide);
      if (!video) { return; }
      slide.setAttribute('data-playing', '');
      video.controls = true;
      // A load that failed earlier (e.g. the server dropped the request) leaves the element in an
      // error state where play() rejects at once, so the button would be dead: reload, then play.
      if (video.error) { video.load(); }
      var started = video.play();
      if (started && started.catch) { started.catch(function () { stop(slide); }); }
    };
    slides.forEach(function (slide, i) {
      var button = slide.querySelector('.reel-play');
      var video = videoOf(slide);
      if (!button || !video) { return; }
      button.addEventListener('click', function () {
        var state = slide.getAttribute('data-kui-step-state');
        if (state && state !== 'active' && dots[i]) { dots[i].click(); }
        play(slide);
      });
      video.addEventListener('ended', function () { stop(slide); video.load(); });
      video.addEventListener('error', function () { stop(slide); });
    });
    if (window.MutationObserver) {
      new MutationObserver(function (records) {
        records.forEach(function (record) {
          if (record.target.getAttribute('data-kui-step-state') !== 'active') { stop(record.target); }
        });
      }).observe(deck, { subtree: true, attributes: true, attributeFilter: ['data-kui-step-state'] });
    }
    if (window.IntersectionObserver) {
      new IntersectionObserver(function (entries) {
        if (!entries[0].isIntersecting) { slides.forEach(stop); }
      }).observe(deck);
    }
  })();
