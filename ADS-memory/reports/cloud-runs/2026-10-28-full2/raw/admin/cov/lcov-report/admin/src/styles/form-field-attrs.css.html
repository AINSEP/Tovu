/**
 * `FieldAttributesDialog` (`FormEditor.tsx`) — the per-field "CSS classes + HTML attributes" modal
 * reached from the field table's kebab trigger. A separate partial rather than an addition to
 * `styles.css` because another pass currently owns that file; same scoped-partial precedent
 * `styles/select.css`/`styles/assistant.css` establish (imported directly by the component that
 * uses it — see `Select.tsx`'s own `import` statement — rather than wired through `main.tsx`).
 *
 * `.settings-dialog-backdrop` (`styles.css`) is `display: grid; place-items: center`, which sizes
 * its child to shrink-to-fit content, so this dialog needs its own `width` (not just `max-width`,
 * which only ever caps a box, never gives it a minimum) to render at a predictable size regardless
 * of how many attribute rows are open.
 *
 * BUG, found live via `getComputedStyle` (not by eye) and fixed here: a bare `.field-attrs-dialog
 * { width: 32rem }` DID NOT WIN. The element carries `className="settings-dialog
 * field-attrs-dialog"`, and `.settings-dialog` (`styles.css`) already sets `max-width: 26rem`
 * (416px) at the same (0,1,0) specificity as a bare `.field-attrs-dialog` rule — at equal
 * specificity the LAST rule in cascade order wins, and this component's CSS module import running
 * after `main.tsx`'s `import "./styles.css"` in source does NOT guarantee `styles.css`'s rule
 * loses: measured on the running (Vite dev) app, `.settings-dialog`'s 26rem was winning and this
 * rule's `width: 32rem` was being clamped down to 416px by that inherited `max-width`, every time,
 * not intermittently. Whatever determines the two stylesheets' relative order in the actual
 * injected `<style>` tags, it is not "which file a component imports" — do not rely on it.
 * Fixed by raising specificity instead of hoping for cascade order: `.settings-dialog.field-
 * attrs-dialog` (two classes, (0,2,0)) strictly outranks `.settings-dialog` alone regardless of
 * source order, no `!important` needed. Re-confirmed after the fix via the same `getComputedStyle`
 * check: `width` 512px, `max-width` matches `calc(100vw - 2rem)` (evaluates below 512px only on a
 * viewport narrower than ~544px).
 */
/* 36rem (was 32rem) since the dialog also holds the per-type settings rows
   (`form-field-settings.css`): an option row is label + value + ↑ ↓ × on one line. */
.settings-dialog.field-attrs-dialog {
  width: 36rem;
  max-width: calc(100vw - 2rem);
  max-height: 85vh;
  overflow-y: auto;
}
 
/* Save/Cancel stay in view while the settings scroll (Chrome check, 2026-10-11: on a long option
   list they scrolled out of the dialog). Sticky to the dialog's own scroll box; the negative bottom
   and margin cancel the dialog's 1.5rem bottom padding so no scrolled content shows below the bar. */
.settings-dialog.field-attrs-dialog .editor-actions {
  position: sticky;
  bottom: -1.5rem;
  z-index: 1;
  display: flex;
  gap: var(--space-2);
  margin-bottom: -1.5rem;
  padding: var(--space-3) 0 1.5rem;
  background: var(--surface);
  border-top: 1px solid var(--border);
}
 
.field-attrs-rows {
  display: flex;
  flex-direction: column;
  gap: var(--space-2);
}
/* The trailing "Add attribute" button is a direct child of the flex column above, whose default
   `align-items: stretch` blew it edge-to-edge (user report: "too wide. left align it and make it a
   standard smaller size"). This is the third live instance of the same stretch-by-default bug in
   this admin — `styles.css`'s `.menu-tree > .btn-secondary` records the identical report, in the
   user's own words, about "+ Add item", and `.settings-scope-editor .btn-danger` is a fourth.
   `align-self: flex-start` sizes it to its own content, which handles both halves of the report
   without a `max-width` magic number.
 
   The size step down is deliberate and scoped here rather than added to shared `.btn-secondary`:
   this button is a *tertiary* action inside a dialog (it adds a row to a list that is itself
   optional), sitting below two real dialog actions. `--text-xs` + the tighter padding is the same
   proportion the `.field-attrs-rows` code chips above already use, so the block reads as one
   scale. Scoped to a direct child so the per-row "Remove" buttons — which live inside
   `.collections-field-row` fieldsets, not directly here — keep the standard button metrics and
   stay visually aligned with their row's inputs. */
.field-attrs-rows > .btn-secondary {
  align-self: flex-start;
  padding: 0.35rem 0.6rem;
  font-size: var(--text-xs);
}
 
/* Each attribute row's Name/Value fields sit side by side, not stacked full-width —
   `.collections-field-row` (`styles.css`) already sets `flex-wrap: wrap` so they still stack at
   narrow widths, this just gives each `.field` room to grow instead of shrinking to its input's
   own intrinsic width, the default for a flex child with no basis. */
.field-attrs-rows .collections-field-row .field {
  flex: 1 1 10rem;
  min-width: 8rem;
}
