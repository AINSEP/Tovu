export type OwnedPlan<T extends object> = T & { readonly planId: string; readonly principalId: string; readonly workspaceId: string; readonly expiresAtMs: number };
export type TakeOwnedPlanResult<T extends object> = { ok: true; plan: OwnedPlan<T> } | { ok: false; code: "PLAN_NOT_FOUND" | "PLAN_EXPIRED" };
export class OwnedPlanStore<T extends object> {

  take(required: { planId: string; principalId: string; workspaceId: string }, optional?: Record<string, never>): TakeOwnedPlanResult<T>;
}
