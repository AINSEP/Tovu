// A0 baseline capture (admin-ui-extraction sheet A0). Run through the journeys config
// (development/playwright.journeys.config.ts): it boots the hermetic seeded site and the
// admin dev server, logs in once and hands this spec an authenticated `page`.
//
// Output: <A0_OUT>/<panelId>__<view>__<theme>-<scheme>-<width>.png
// Request counts: <A0_OUT>/request-counts-${width}.json (GET/other requests per screen open, classic-light-1440).
//
// Views per panel: `list` (the panel's own route), `tab-<n>-<label>` for each [role=tab] found at
// runtime, and `detail` (first row link in the main region) when one exists.
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import { test, expect, type Page } from "@playwright/test";
import { ADDONS_PANEL_PATHS } from "../support/addons-route-fixture.js";

const OUT = process.env.A0_OUT ?? path.resolve("development/test-results/a0");
const PANELS_SRC = path.resolve(import.meta.dirname, "../../../apps/admin/src/panels.tsx");
const PANEL_IDS = [...readFileSync(PANELS_SRC, "utf8").matchAll(/^\s+id: ["']([a-z0-9-]+)["']/gm)].map((m) => m[1]!);
const THEMES = ["classic", "glass"] as const;
const SCHEMES = ["light", "dark"] as const;
const WIDTHS = [1440, 390] as const;
const ONLY = process.env.A0_ONLY?.split(",").filter(Boolean);
const FREEZE = `*,*::before,*::after{animation:none!important;transition:none!important;caret-color:transparent!important}`;

mkdirSync(OUT, { recursive: true });

async function applyAppearance(page: Page, { theme, scheme }: { theme: string; scheme: string }) {
  await page.evaluate(({ theme, scheme }) => {
    const root = document.documentElement;
    root.setAttribute("data-admin-theme", theme);
    root.setAttribute("data-color-scheme", scheme);
    if (scheme === "dark") root.setAttribute("data-theme", "dark"); else root.removeAttribute("data-theme");
  }, { theme, scheme });
}

async function settle(page: Page) {
  await page.waitForLoadState("domcontentloaded");
  // The settings feed keeps the network busy, so "networkidle" never fires: wait for the heading, then a fixed beat.
  await page.locator("main h1, h1").first().waitFor({ timeout: 6000 }).catch(() => undefined);
  await page.waitForTimeout(700);
}

async function shoot(page: Page, name: string) {
  await page.screenshot({ path: path.join(OUT, `${name}.png`), fullPage: false });
}

const slug = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "").slice(0, 32) || "tab";

test.describe.configure({ mode: "parallel" });
test.setTimeout(30 * 60_000);

for (const width of WIDTHS) test(`A0 capture ${width}`, async ({ page }) => {
  const counts: Record<string, { total: number; byMethod: Record<string, number> }> = {};
  const manifest: string[] = [];
  const landed: string[] = [];
  {
    await page.setViewportSize({ width, height: width === 390 ? 844 : 900 });
    for (const id of PANEL_IDS) {
      if (ONLY && !ONLY.includes(id)) continue;
      const route = id === "dashboard" ? "/admin/" : `/admin${ADDONS_PANEL_PATHS[id] ?? `/${id}`}`;
      const reqs: string[] = [];
      const onReq = (r: { method(): string; url(): string }) => { if (r.url().includes("/api/")) reqs.push(r.method()); };
      page.on("request", onReq);
      await page.goto(route, { waitUntil: "domcontentloaded" }).catch(() => undefined);
      await page.addStyleTag({ content: FREEZE }).catch(() => undefined);
      await settle(page);
      page.off("request", onReq);
      if (width === 1440) {
        const byMethod: Record<string, number> = {};
        for (const m of reqs) byMethod[m] = (byMethod[m] ?? 0) + 1;
        counts[id] = { total: reqs.length, byMethod };
      }
      const tabs = await page.locator('main [role="tab"]').evaluateAll((els) => els.map((el, i) => ({ i, label: (el.textContent ?? "").trim() })));
      const views: { name: string; act: (p: Page) => Promise<void> }[] = [{ name: "list", act: async () => undefined }];
      tabs.slice(0, 8).forEach(({ i, label }) => views.push({
        name: `tab-${i + 1}-${slug(label)}`,
        act: async (p) => { await p.locator('main [role="tab"]').nth(i).click({ timeout: 4000 }); await p.waitForTimeout(600); },
      }));
      for (const v of views) {
        if (v.name !== "list") {
          await page.goto(route, { waitUntil: "domcontentloaded" }).catch(() => undefined);
          await page.addStyleTag({ content: FREEZE }).catch(() => undefined);
          await settle(page);
        }
        await v.act(page).catch(() => undefined);
        for (const theme of THEMES) for (const scheme of SCHEMES) {
          await applyAppearance(page, { theme, scheme });
          await page.waitForTimeout(120);
          const name = `${id}__${v.name}__${theme}-${scheme}-${width}`;
          if (theme === "classic" && scheme === "light") {
            const h1 = await page.locator("h1").first().textContent({ timeout: 1500 }).catch(() => "");
            landed.push(`${name}\t${new URL(page.url()).pathname}${new URL(page.url()).search}\t${(h1 ?? "").trim().slice(0, 50)}`);
          }
          await shoot(page, name);
          manifest.push(name);
        }
      }
    }
  }
  writeFileSync(path.join(OUT, `request-counts-${width}.json`), JSON.stringify(counts, null, 2));
  writeFileSync(path.join(OUT, `landed-${width}.tsv`), landed.join("\n") + "\n");
  writeFileSync(path.join(OUT, `manifest-${width}.txt`), manifest.join("\n") + "\n");
  expect(manifest.length).toBeGreaterThan(0);
});
