#!/usr/bin/env bash
# usage: p10a-boot.sh <tree> <site> <port> <label>   (HOME already exported by caller)
tree=$1; site=$2; port=$3; label=$4; out=$S/raw/p10a-$label-serve.txt
cd $tree
TSX_TSCONFIG_PATH=apps/site-chat/tsconfig.json setsid node --import tsx apps/website/src/cli/main.ts serve "$site" --port $port > $out 2>&1 &
pid=$!; echo $pid > $S/p10a/$label.pid
code=000; for i in $(seq 1 120); do code=$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:$port/ 2>/dev/null); [ "$code" = 200 ] && break; sleep 1; done
echo "first-200 after ${i}s code=$code" >> $out
sleep 8
kill -TERM -$pid 2>/dev/null; for i in $(seq 1 20); do kill -0 $pid 2>/dev/null || break; sleep 0.5; done
kill -KILL -$pid 2>/dev/null; wait $pid 2>/dev/null
kill -0 $pid 2>/dev/null && echo "STILL ALIVE $pid" >> $out || echo "stopped pid $pid" >> $out
echo "code=$code"
